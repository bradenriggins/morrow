# Morrow for Muse: packaging deny-list.
#
# Anything matching the [paths] section must NEVER appear in the carved
# distribution tree. The carve script enforces a PINNED copy of this file
# (kept next to the carve script, outside the tree) and fails loudly if
# the tree's copy differs from the pinned one, so weakening this file in
# the tree can never weaken the carve gate. scripts/verify-no-secrets.sh
# enforces every section against any tree it is pointed at (default: the
# tree it ships in); install.sh runs it as the last install step.
#
# Matching rules for [paths]:
#   * Patterns match the tree-relative path with '/' separators.
#   * A pattern ending in '/' is a directory: the directory itself and
#     everything under it is denied.
#   * Otherwise the pattern is a shell glob ('*' spans directories).
#   * Matching is CASE-INSENSITIVE: the path and the pattern are both
#     lowercased before matching, so COOKIES-JOURNAL and cookies match
#     the Cookies* pattern.
#   * Before matching, the path is Unicode-normalized (NFKC), invisible
#     characters (zero-width spaces, soft hyphens, BOM) are stripped, and
#     common confusable lookalikes (Cyrillic/Greek homoglyphs of ASCII
#     letters) are folded to ASCII. A homoglyph variant of a denied name
#     (e.g. a Cyrillic-A "Cookies") is caught, not evaded.
#   * The gate never scans itself: pack/deny-list.txt and
#     scripts/verify-no-secrets.sh are skipped by every check.
#
# Matching rules for [content]:
#   * One extended regular expression per line, checked against every text
#     file in the tree. Binary files are skipped.
#   * Files in [content_exempt] skip the [content] check AND the tenant
#     rule: they are proof-evidence files carrying the PROOF-EVIDENCE
#     banner, quoting sanitized example output. Never add an exemption
#     without that banner.
#   * The gate never scans itself: pack/deny-list.txt and
#     scripts/verify-no-secrets.sh are skipped by the [content] check.
#
# TENANT RULE (enforced by verify-no-secrets.sh, not a plain pattern):
# the only instructure.com hostnames allowed in the tree are the ones in
# [tenant_allow]. Hostnames are extracted case-insensitively and compared
# in lowercase, so https://EVIL.INSTRUCTURE.COM/ cannot evade the rule.
# Any other *.instructure.com hostname is an account-specific default and
# fails the gate, unless the file is in [content_exempt].

[content]
# Live session cookie material: assignments, not mere mentions.
# Both the =-assignment form (URL/query) and the JSON object-literal
# form ({"canvas_session": "..."}) are denied: the JSON form evaded the
# =-only patterns before.
canvas_session=[A-Za-z0-9%._~+-]{16,}
"canvas_session"\s*:\s*"[^"]{16,}"
'canvas_session'\s*:\s*'[^']{16,}'
remember_user_token=[^;"' ]{16,}
"remember_user_token"\s*:\s*"[^"]{16,}"
'remember_user_token'\s*:\s*'[^']{16,}'
# CSRF token material: the runtime capture lane reads this from the live
# page context and writes it to ~/.morrow/session.json (outside any tree).
# A literal assignment here is a leak: the same value grants write access.
_csrf_token\s*=\s*[^;"' ]{16,}|"_csrf_token"\s*:\s*"[^"]{16,}"
_legacy_normandy_session
pseudonym_credentials
# Private key material.
-----BEGIN (RSA )?PRIVATE KEY-----
-----BEGIN OPENSSH PRIVATE KEY-----
# SPKI pin literals: pins are derived from the environment at launch,
# never written into the tree.
--ignore-certificate-errors-spki-list=[^ "']{20,}
# Authorization header with a literal credential value (not a placeholder
# name, handle, or redacted marker): a real token in this position is
# shipped auth material.
"Authorization"\s*:\s*"[A-Za-z0-9_\-.~+/=]{20,}
'Authorization'\s*:\s*'[A-Za-z0-9_\-.~+/=]{20,}
# Proxy credentials: the egress probe redacts them; a literal userinfo
# in a URL is a leak. Format placeholders (%s) are not credentials.
https?://[^/ '"@%]+:[^/ '"@%]+@
